**Goal:** A Streamlit app showing a symbol's volume profile, a live VWAP order's execution against the schedule that profile implies, and today's realised volume — on one shared time axis.

**Architecture:** Four pure DataFrame-in/DataFrame-out modules (`session`, `profile`, `schedule`, `realized`) under a `DataProvider` ABC with a demo and a pykx implementation. Streamlit and plotly sit only in `ui/` and `app.py`. Nothing below `ui/` imports streamlit, so the whole decoding layer is testable with no kdb and no pykx.

**Tech Stack:** Python 3.13, streamlit 1.45, plotly 5.24, pandas 2.2, pytest. pykx optional and imported lazily.

**Spec:** `core-vp/docs/superpowers/specs/2026-09-04-core-vp-design.md`

## Global Constraints

- Working directory for every path below is `core-vp/`.
- No module under `core/` may import `streamlit` or `plotly`. Tests enforce this.
- `pykx` is imported lazily inside functions, never at module scope. The app must run with it absent.
- Header-row constants come from `idxprof.q:37-42`: `time` 0/1/2/3 = adv / open auction / close auction / PM reopen auction.
- `MIN_BUCKET_MINUTE = 100` discriminates header rows from minutes-of-day.
- `vtot = adv + v_open + v_close + v_open_pm` — includes `v_open_pm`, deliberately unlike `getsymprof`. See spec §1.2.
- Money/qty formatting: thousands separators, no decimals. Fractions as percentages with 2 decimals.
- Demo data uses the real values photographed for `000001.C2`: `adv=105156029`, `v_open=445700`, `v_close=799800`, `v_open_pm=2000`.

```python
# core/session.py
"""Session boundaries, derived from the profile's own buckets.

No per-market table: the lunch break is the largest gap between consecutive
bucket minutes that exceeds twice the median spacing. China's 690 -> 780 is a
90-minute gap against a 10-minute median; a continuous market has no such gap.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Sequence

import statistics

LUNCH_GAP_FACTOR = 2.0


@dataclass(frozen=True)
class Session:
    open_min: int
    close_min: int
    lunch: tuple[int, int] | None
    has_pm_auction: bool
    warnings: tuple[str, ...] = field(default=())

    @property
    def is_two_session(self) -> bool:
        return self.lunch is not None


def derive_session(minutes: Sequence[int], v_open_pm: float) -> Session:
    m = sorted(int(x) for x in minutes)
    if len(m) < 2:
        raise ValueError(f"need at least 2 buckets to derive a session, got {len(m)}")

    gaps = [b - a for a, b in zip(m, m[1:])]
    median_gap = statistics.median(gaps)
    widest = max(gaps)
    lunch = None
    if median_gap > 0 and widest > LUNCH_GAP_FACTOR * median_gap:
        i = gaps.index(widest)
        lunch = (m[i], m[i + 1])

    has_pm = float(v_open_pm) > 0.0
    warnings: list[str] = []
    if has_pm and lunch is None:
        warnings.append(
            "profile reports a PM reopen auction (time=3) but no lunch break was "
            "found in the buckets - the data disagrees with itself"
        )
    return Session(open_min=m[0], close_min=m[-1], lunch=lunch,
                   has_pm_auction=has_pm, warnings=tuple(warnings))
```

```python
# core/profile.py
"""Decode the VPROF `vst` rows into a usable profile.

Row layout, per idxprof.q:37-42:
    time=0  adv        median daily continuous volume
    time=1  vopen      open auction volume
    time=2  vclose     close auction volume
    time=3  vopenpm    PM reopen auction volume (lunch-break markets)
    time>=100          minutes-of-day; vmed is a CUMULATIVE fraction

getsymprof takes `deltas` of vmed, which is what establishes it as cumulative.
The header-row count lives in cc0 on row 0 (load_equity_special.q:128); the
fixed 10/11 that getsymprof and idxprof hard-code does not survive a symbol
with a different number of pre-open rows.
"""
from __future__ import annotations

import datetime as dt
from dataclasses import dataclass

import numpy as np
import pandas as pd

from core.session import Session, derive_session

MIN_BUCKET_MINUTE = 100
CUM_TOLERANCE = 0.01

ROW_ADV, ROW_OPEN, ROW_CLOSE, ROW_OPEN_PM = 0, 1, 2, 3

PHASE_OPEN = "open auction"
PHASE_CONTINUOUS = "continuous"
PHASE_PM = "pm auction"
PHASE_CLOSE = "close auction"


class ProfileError(ValueError):
    """The profile rows cannot be decoded into a usable curve."""


@dataclass(frozen=True)
class Profile:
    sym: str
    date: dt.date
    adv: float
    v_open: float
    v_close: float
    v_open_pm: float
    buckets: pd.DataFrame
    session: Session

    @property
    def vtot(self) -> float:
        return self.adv + self.v_open + self.v_close + self.v_open_pm

    @property
    def continuous(self) -> pd.DataFrame:
        return self.buckets[self.buckets.phase == PHASE_CONTINUOUS]


def _clock(minute: int) -> str:
    return f"{int(minute) // 60:02d}:{int(minute) % 60:02d}"


def _header_skip(raw: pd.DataFrame) -> int:
    if "cc0" in raw.columns and len(raw):
        try:
            n = int(raw["cc0"].iloc[0])
            if 0 < n < len(raw):
                return n
        except (TypeError, ValueError):
            pass
    mask = raw["time"] >= MIN_BUCKET_MINUTE
    if not mask.any():
        raise ProfileError("no bucket rows: every `time` is below "
                           f"{MIN_BUCKET_MINUTE}")
    return int(np.argmax(mask.to_numpy()))


def decode_profile(raw: pd.DataFrame, sym: str, date: dt.date) -> Profile:
    if raw is None or len(raw) == 0:
        raise ProfileError(f"no rows returned for {sym} on {date}")

    raw = raw.sort_values("time").reset_index(drop=True)
    head = raw[raw["time"] < MIN_BUCKET_MINUTE].set_index("time")["vmed"]
    missing = [r for r in (ROW_ADV, ROW_OPEN, ROW_CLOSE) if r not in head.index]
    if missing:
        raise ProfileError(f"missing header row(s) {missing} for {sym} on {date}; "
                           f"expected time=0/1/2 to carry adv/open/close")

    adv = float(head.get(ROW_ADV, 0.0))
    v_open = float(head.get(ROW_OPEN, 0.0))
    v_close = float(head.get(ROW_CLOSE, 0.0))
    v_open_pm = float(head.get(ROW_OPEN_PM, 0.0))

    body = raw.iloc[_header_skip(raw):]
    body = body[body["time"] >= MIN_BUCKET_MINUTE]
    body = body[body["vmed"] > 0.0]
    if len(body) < 2:
        raise ProfileError(f"only {len(body)} non-zero bucket(s) for {sym} on "
                           f"{date}; cannot form a curve")

    minutes = body["time"].astype("int64").tolist()
    cum = body["vmed"].astype(float).to_numpy()

    if np.any(np.diff(cum) < -1e-12):
        raise ProfileError(f"vmed is not non-decreasing for {sym} on {date}; "
                           "it is read as a cumulative fraction")
    if abs(cum[-1] - 1.0) > CUM_TOLERANCE:
        raise ProfileError(f"vmed does not reach 1.0 for {sym} on {date} "
                           f"(ends at {cum[-1]:.4f})")

    session = derive_session(minutes, v_open_pm)

    share = np.diff(cum, prepend=cum[0])
    share[0] = 0.0
    cont = pd.DataFrame({
        "minute": minutes,
        "phase": PHASE_CONTINUOUS,
        "cum_frac": cum,
        "share": share,
    })

    vtot = adv + v_open + v_close + v_open_pm
    if vtot <= 0:
        raise ProfileError(f"total volume is {vtot} for {sym} on {date}")
    cont["share_of_day"] = cont["share"] * (adv / vtot)

    step = int(np.median(np.diff(sorted(set(minutes))))) or 1
    extras = [{"minute": session.open_min - step, "phase": PHASE_OPEN,
               "cum_frac": np.nan, "share": np.nan,
               "share_of_day": v_open / vtot}]
    if session.lunch is not None:
        extras.append({"minute": session.lunch[1] - step, "phase": PHASE_PM,
                       "cum_frac": np.nan, "share": np.nan,
                       "share_of_day": v_open_pm / vtot})
    extras.append({"minute": session.close_min + step, "phase": PHASE_CLOSE,
                   "cum_frac": np.nan, "share": np.nan,
                   "share_of_day": v_close / vtot})

    buckets = pd.concat([cont, pd.DataFrame(extras)], ignore_index=True)
    buckets = buckets.sort_values("minute").reset_index(drop=True)
    buckets["cum_of_day"] = buckets["share_of_day"].cumsum()
    buckets["clock"] = buckets["minute"].map(_clock)
    buckets = buckets[["minute", "clock", "phase", "cum_frac", "share",
                       "share_of_day", "cum_of_day"]]

    return Profile(sym=sym, date=date, adv=adv, v_open=v_open, v_close=v_close,
                   v_open_pm=v_open_pm, buckets=buckets, session=session)
```

```python
# core/schedule.py
"""The order's expected curve, and what it actually did.

Three actual series, because two of them are routinely confused:

    filled                 execution.cum_qty            where the order is
    committed              target_state.commit_open
                           + commit_close               reserved for the auctions
    filled_plus_committed  the sum                      including what it holds back

A VWAP order reserving quantity for the close reads as behind schedule on a
naive filled-vs-expected chart while doing exactly what it was told. The gap
between the second and third series is that reserve, made visible.
"""
from __future__ import annotations

import datetime as dt

import numpy as np
import pandas as pd

from core.profile import PHASE_CLOSE, PHASE_OPEN, Profile


def to_minute(value) -> int | None:
    """Minutes-of-day from whatever kdb handed back, or None."""
    if value is None or value is pd.NaT:
        return None
    if isinstance(value, pd.Timedelta):
        return int(value.total_seconds() // 60)
    if isinstance(value, dt.time):
        return value.hour * 60 + value.minute
    if isinstance(value, dt.timedelta):
        return int(value.total_seconds() // 60)
    try:
        if pd.isna(value):
            return None
    except (TypeError, ValueError):
        pass
    try:
        return int(value)
    except (TypeError, ValueError):
        return None


def _step_series(frame: pd.DataFrame, time_col: str, value_cols: list[str],
                 minutes: list[int]) -> pd.DataFrame:
    """Last value at or before each grid minute, held flat in between."""
    out = pd.DataFrame({c: np.zeros(len(minutes), dtype=float) for c in value_cols})
    if frame is None or len(frame) == 0:
        return out
    f = frame.copy()
    f["_m"] = f[time_col].map(to_minute)
    f = f.dropna(subset=["_m"]).sort_values("_m")
    if len(f) == 0:
        return out
    grouped = f.groupby("_m")[value_cols].last()
    idx = grouped.index.to_numpy()
    for c in value_cols:
        vals = grouped[c].astype(float).to_numpy()
        pos = np.searchsorted(idx, np.array(minutes), side="right") - 1
        out[c] = np.where(pos >= 0, vals[np.clip(pos, 0, len(vals) - 1)], 0.0)
    return out


def build_schedule(profile: Profile, order: pd.Series,
                   executions: pd.DataFrame, states: pd.DataFrame) -> pd.DataFrame:
    size = float(order.get("size", 0) or 0)

    b = profile.buckets.copy()
    if not int(order.get("doopen", 1) or 0):
        b = b[b.phase != PHASE_OPEN]
    if not int(order.get("doclose", 1) or 0):
        b = b[b.phase != PHASE_CLOSE]

    lo = to_minute(order.get("t_start"))
    hi = to_minute(order.get("t_end"))
    if lo is not None:
        b = b[b.minute >= lo]
    if hi is not None:
        b = b[b.minute <= hi]
    if len(b) == 0:
        raise ValueError("the order's window selects no profile buckets")

    b = b.sort_values("minute").reset_index(drop=True)
    total = b["share_of_day"].sum()
    if total <= 0:
        raise ValueError("the order's window carries no expected volume")
    b["expected_qty"] = (b["share_of_day"] / total).cumsum() * size

    minutes = b["minute"].astype(int).tolist()
    filled = _step_series(executions, "time", ["cum_qty"], minutes)["cum_qty"]
    commit = _step_series(states, "t_algo", ["commit_open", "commit_close"], minutes)

    b["filled_qty"] = filled.to_numpy()
    b["committed_qty"] = (commit["commit_open"] + commit["commit_close"]).to_numpy()
    b["filled_plus_committed"] = b["filled_qty"] + b["committed_qty"]
    b["ahead_qty"] = b["filled_qty"] - b["expected_qty"]

    return b[["minute", "clock", "phase", "expected_qty", "filled_qty",
              "committed_qty", "filled_plus_committed", "ahead_qty"]]
```

```python
# core/realized.py
"""Today's realised volume, bucketed onto the profile's own grid.

Follows getsymprof_realRT: a trade belongs to the first grid minute at or after
it, and the opening auction print is subtracted from whichever bucket contains
it and seated in the first bucket instead, so it is not counted twice into the
continuous curve.
"""
from __future__ import annotations

import numpy as np
import pandas as pd

from core.profile import Profile
from core.schedule import to_minute

TIME_COLUMNS = ("tradeTime", "time", "t_algo")


def _time_column(qatt: pd.DataFrame) -> str | None:
    for c in TIME_COLUMNS:
        if c in qatt.columns:
            return c
    return None


def realized_curve(qatt: pd.DataFrame, profile: Profile,
                   open_print: tuple[int, float] | None = None) -> pd.DataFrame:
    grid = profile.continuous["minute"].astype(int).to_numpy()
    clock = profile.continuous["clock"].to_numpy()
    volume = np.zeros(len(grid), dtype=float)

    col = _time_column(qatt) if qatt is not None and len(qatt) else None
    if col is not None:
        f = qatt[[col, "size"]].copy()
        f["_m"] = f[col].map(to_minute)
        f = f.dropna(subset=["_m", "size"])
        f = f[f["size"] > 0]
        if len(f):
            pos = np.searchsorted(grid, f["_m"].to_numpy(), side="left")
            keep = pos < len(grid)
            np.add.at(volume, pos[keep], f["size"].to_numpy()[keep])

    if open_print is not None:
        op_minute, op_size = open_print
        pos = int(np.searchsorted(grid, op_minute, side="left"))
        if pos < len(grid):
            volume[pos] -= float(op_size)
        volume[0] = float(op_size)
        volume = np.maximum(volume, 0.0)

    cum = volume.cumsum()
    total = cum[-1] if len(cum) else 0.0
    return pd.DataFrame({
        "minute": grid,
        "clock": clock,
        "volume": volume,
        "cum_volume": cum,
        "share": volume / total if total > 0 else np.nan,
        "cum_frac": cum / total if total > 0 else np.nan,
    })
```
